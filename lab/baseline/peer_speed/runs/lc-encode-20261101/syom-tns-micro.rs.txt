use std::hint::black_box;
use std::time::Instant;
mod engine {
    pub mod swb { pub const LONG_WINDOW_LEN: usize=1024; }
    pub mod tns {
        const MAX_ORDER: usize=12;
        #[path="/tmp/syom-tns-micro-base.rs"] mod base;
        #[path="/tmp/syom-tns-micro-current.rs"] mod current;
        pub fn correlation(which: bool,x:&[f32])->[f64;13] {
            if which { current::autocorrelate(x) } else { base::autocorrelate(x) }
        }
        pub fn filter(which:bool,x:&mut[f32],original:&[f32],a:&[f64]) {
            if which { current::analysis_filter(x,original,a) } else { base::analysis_filter(x,original,a) }
        }
    }
}
fn main() {
 let data: Vec<f32>=(0..1024).map(|i| ((i*37%101) as f32-50.0)*0.01).collect();
 let a:Vec<f64>=(0..12).map(|i|(i as f64-5.0)*0.13).collect();
 let mut output=vec![0.0f32;1024];
 for kind in ["correlation","filter"] {
  let grids:&[(usize,usize)]=if kind=="correlation" { &[(32,0),(128,0),(512,0),(1024,0)] } else { &[(512,2),(512,6),(512,12)] };
  for &(n,order) in grids {
   for (round,which) in [false,true,true,false].into_iter().enumerate() {
    let mut run=|| { if kind=="correlation" { black_box(engine::tns::correlation(black_box(which),black_box(&data[..n]))); } else { engine::tns::filter(black_box(which),black_box(&mut output[..n]),black_box(&data[..n]),black_box(&a[..order])); black_box(&output); } };
    for _ in 0..100 { run(); }
    let mut samples=Vec::new();
    for _ in 0..11 {
     let start=Instant::now();
     for _ in 0..2000 { run(); }
     samples.push(start.elapsed().as_secs_f64()*1e9/2000.0);
    }
    samples.sort_by(f64::total_cmp);
    println!("{{\"kind\":\"{kind}\",\"n\":{n},\"order\":{order},\"round\":{round},\"candidate\":{which},\"median_ns\":{},\"samples_ns\":{:?}}}",samples[5],samples);
   }
  }
 }
}
