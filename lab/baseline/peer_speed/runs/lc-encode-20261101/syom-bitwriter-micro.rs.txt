use std::hint::black_box;
use std::time::Instant;
mod engine {
 #[path="/home/ekhodzitsky/Projects/syom/src/engine/error.rs"] mod error;
 #[path="/tmp/syom-bitwriter-micro-base.rs"] pub mod base;
 #[path="/tmp/syom-bitwriter-micro-current.rs"] pub mod current;
}
macro_rules! emit {
 ($name:ident,$writer:ty)=>{
  fn $name(fields:&[(u32,u32)],flags:bool,buffer:Vec<u8>)->(Vec<u8>,u64) {
   let mut writer=<$writer>::from_vec(buffer);
   writer.write(0x55,7);
   for &(value,width) in fields {
    if flags { writer.write_bit(value&1!=0); } else { writer.write(value,width); }
   }
   let bits=writer.bit_len();
   (writer.finish(),bits)
  }
 }
}
emit!(base,engine::base::BitWriter);
emit!(current,engine::current::BitWriter);
fn main(){
 let cases:&[(&str,&[u32],bool)]=&[
  ("huffman-mixed",&[1,3,5,8,12,4,9,2,1,16,7,6],false),
  ("byte-fill",&[8],false),
  ("sign-flags",&[1],true),
  ("header-wide",&[32,31,24,13,11,8,0,3,5],false),
 ];
 for &(label,widths,flags) in cases {
  let mut seed=0xC0FFEE42u32;
  let fields:Vec<_>=(0..1000).map(|i|{seed=seed.wrapping_mul(1664525).wrapping_add(1013904223);(seed,widths[i%widths.len()])}).collect();
  let reference=base(&fields,flags,Vec::new());
  assert_eq!(reference,current(&fields,flags,Vec::new()));
  for (round,candidate) in [false,true,true,false].into_iter().enumerate() {
   let mut buffer=Vec::with_capacity(8192);
   let mut run=||{let out=if black_box(candidate){current(black_box(&fields),black_box(flags),std::mem::take(&mut buffer))}else{base(black_box(&fields),black_box(flags),std::mem::take(&mut buffer))};black_box(out.1);buffer=out.0;black_box(&buffer);};
   for _ in 0..100 {run();}
   let mut samples=Vec::new();
   for _ in 0..11 {let begin=Instant::now();for _ in 0..2000 {run();} samples.push(begin.elapsed().as_secs_f64()*1e9/2000.0);}
   samples.sort_by(f64::total_cmp);
   println!("{{\"kind\":\"{label}\",\"fields\":1000,\"prefix_bits\":7,\"output_bytes\":{},\"round\":{round},\"candidate\":{candidate},\"median_ns\":{},\"samples_ns\":{:?}}}",reference.0.len(),samples[5],samples);
  }
 }
}
